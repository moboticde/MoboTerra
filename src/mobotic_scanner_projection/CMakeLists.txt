cmake_minimum_required(VERSION 3.13)
project(mobotic_scanner_projection)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sick_safetyscanners2_interfaces REQUIRED)
add_executable(scanner_evaluation_projection src/scanner_evaluation_projection.cpp)
target_include_directories(scanner_evaluation_projection PRIVATE include)
ament_target_dependencies(scanner_evaluation_projection rclcpp sick_safetyscanners2_interfaces)
install(TARGETS scanner_evaluation_projection DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_test REQUIRED)
  add_executable(test_projection test/test_projection.cpp)
  target_include_directories(test_projection PRIVATE include)
  ament_target_dependencies(test_projection sick_safetyscanners2_interfaces)
  ament_add_test(test_projection COMMAND "$<TARGET_FILE:test_projection>" GENERATE_RESULT_FOR_RETURN_CODE_ZERO)
endif()
ament_package()
